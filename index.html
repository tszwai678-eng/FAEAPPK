function updateAll(){ const inputs = document.querySelectorAll("input[type=number]"); let updates = {}; inputs.forEach(input=>{ const name = input.dataset.name; const value = parseFloat(input.value); if(!isNaN(value) && value !== 0){ const currentTotal = parseFloat(input.dataset.total) || 0; updates["scores/"+name+"/total"] = currentTotal + value; } }); if(Object.keys(updates).length === 0){ alert("沒有輸入任何分數"); return; } db.ref().update(updates).then(()=>{ document.querySelectorAll("input[type=number]").forEach(i=>i.value=""); alert("✅ 本次分數已全部更新"); }); }
